### 2. HAL层 — sslv_hal.h + sslv_hal.c

职责：把硬件寄存器的位域操作封装成有意义的函数名。不关心数据流逻辑、不关心中断处理。

**寄存器地址**

#define SSI_SLV_IOBASE 0x10045000

static const unsigned long iobase[] = {
    SSI_SLV_IOBASE,    // id=0 的基址
};

// 最终寄存器访问：
#define SSLV_ADDR(id, reg) ((volatile unsigned long *)(KSEG1ADDR(iobase[id]) + reg))
KSEG1ADDR 把物理地址映射到 MIPS uncached 段（0x10045000 → 0xB0045000），确保每次读写都直接到硬件总线，不经过 CPU cache。

**寄存器偏移**

#define CTRLR    0x0000   // 控制寄存器
#define SSIENR   0x0008   // 使能寄存器
#define MWCR     0x000C   // Microwire 控制
#define TXFTLR   0x0018   // TX FIFO 阈值
#define RXFTLR   0x001C   // RX FIFO 阈值
#define TXFLR    0x0020   // TX FIFO 水位 (只读)
#define RXFLR    0x0024   // RX FIFO 水位 (只读)
#define SR       0x0028   // 状态寄存器
#define IMR      0x002C   // 中断屏蔽
#define ISR      0x0030   // 中断状态 (屏蔽后)
#define RISR     0x0034   // 中断状态 (raw，屏蔽前)
#define TXOICR   0x0038   // TX 溢出清中断
#define RXOICR   0x003C   // RX 溢出清中断
#define RXUICR   0x0040   // RX 下溢清中断
#define ICR      0x0048   // 组合清中断
#define DMACR    0x004C   // DMA 控制
#define DMATDLR  0x0050   // DMA TX 数据水位
#define DMARDLR  0x0054   // DMA RX 数据水位
#define IDR      0x0058   // ID 寄存器
#define DR       0x0060   // 数据寄存器 (读写 FIFO 的窗口)
**CTRLR 位域定义**

Bit:  31-23    22-21      20-16      15-12   11     10       9-8     7       6       5-4     3-0
     ┌───────┬─────────┬──────────┬────────┬─────┬────────┬────────┬───────┬───────┬───────┬──────┐
     │  RSVD │ SPI_FRF │  DFS_32  │  CFS   │ SRL │ SLV_OE │  TMOD  │ SCPOL │ SCPH  │  FRF  │ RSVD │
     └───────┴─────────┴──────────┴────────┴─────┴────────┴────────┴───────┴───────┴───────┴──────┘
*对应的位域宏：*
#define CTRLR_SPI_FRF   21, 22    // Motorola SPI 子格式 (需与 FRF 配合)
#define CTRLR_DFS_32    16, 20    // 数据帧长度 (size-1): 7=8bit, 15=16bit, 31=32bit
#define CTRLR_CFS       12, 15    // Microwire 控制字长度
#define CTRLR_SRL       11, 11    // 回环测试
#define CTRLR_SLV_OE    10, 10    // 从机输出使能: 0=DT 输出, 1=DT 高阻
#define CTRLR_TMOD      8, 9      // 传输模式: 00=收发, 01=仅发, 10=仅收
#define CTRLR_SCPOL     7, 7      // 时钟极性: 0=空闲低, 1=空闲高
#define CTRLR_SCPH      6, 6      // 时钟相位: 0=首边沿采样, 1=次边沿采样
#define CTRLR_FRF       4, 5      // 帧格式: 0=SPI, 1=SSP, 2=Microwire

*关键组合：*
Mode 3 (CPOL=1, CPHA=1)：SCPOL=1, SCPH=1
Mode 0 (CPOL=0, CPHA=0)：SCPOL=0, SCPH=0
Motorola SPI：FRF=0, SPI_FRF=0（两个都要设为 0）

**SSIENR 位域**
Bit:  31-1       0
     ┌────────┬──────┐
     │  RSVD  │ SSIE │
     └────────┴──────┘
SSIE=1：使能 SSI_SLV，开始响应 SPI 总线
SSIE=0：立即停止传输，清空 TX/RX FIFO

**SR 状态寄存器**
Bit:  6-31    5      4      3      2      1      0
     ┌─────┬─────┬─────┬─────┬─────┬─────┬──────┐
     │RSVD │ TXE │ RFF │ RFNE│ TFE │ TFNF│ BUSY │
     └─────┴─────┴─────┴─────┴─────┴─────┴──────┘
----------------------------------------
位	名称	            含义
0	BUSY	传输进行中
1	TFNF	TX FIFO 未满
2	TFE	    TX FIFO 空
3	RFNE	RX FIFO 非空
4	RFF	    RX FIFO 满
5	TXE	    传输错误（启动时 TX FIFO 为空）
----------------------------------------

**IMR / ISR / RISR 中断体系**
    IMR  (0x2C):  中断使能屏蔽                   ← 写 1 开中断
    ISR  (0x30):  中断状态 (masked)   ← 中断使能 AND 中断置位 时才为 1
    RISR (0x34):  中断状态 (raw)      ← 硬件中断位，不管是否屏蔽

Bit:  5-31    4        3        2        1        0
     ┌─────┬───────┬───────┬───────┬───────┬───────┐
     │RSVD │ RXFIM │ RXOIM │ RXUIM │ TXOIM │ TXEIM │
     └─────┴───────┴───────┴───────┴───────┴───────┘
             RX阈值     RX溢出      RX下溢     TX溢出     TX空阈值
ISR 是软件真正关心的——只在中断使能且硬件触发时才置位。中断处理中用：
```c
sslv_hal_get_receive_threshold_intr_status(id)  // ISR & IMR 双重判断
sslv_hal_get_receive_overflow_intr_status(id)
```
**基元操作**

// 寄存器读写基元
```c
static inline void sslv_write_reg(id, reg, value)  // *SSLV_ADDR = value
static inline unsigned int sslv_read_reg(id, reg)   // return *SSLV_ADDR
```
// 位域读写基元（调用 bit_field.h 的 set_bit_field / get_bit_field）
```c
static inline void sslv_set_bits(id, reg, start, end, val)   // RMW 写位域
static inline unsigned int sslv_get_bits(id, reg, start, end) // 读位域
```
// FIFO 数据读写
```c
unsigned int sslv_hal_read_fifo(id)   // 读 DR → 弹出 RX FIFO 一条记录
void sslv_hal_write_fifo(id, data)    // 写 DR → 压入 TX FIFO 一条记录
```

## 第二层：中间层 — sslv.h + sslv.c
*职责：管理设备生命周期、中断处理、数据传输流程、ioctl 派发。*

**内核内部数据结构**

struct sslv_config_data (sslv.h:10-30)

struct sslv_config_data {
    unsigned int id;             // 控制器编号，始终 = 0
    char *name;                  // 标识名（未实际使用）
    unsigned int bits_per_word;  // 数据位宽: 8 / 16 / 32
    unsigned int sslv_pol;       // 时钟极性: 0 或 1
    unsigned int sslv_pha;       // 时钟相位: 0 或 1
    unsigned int loop_mode;      // 回环测试模式
};

*全局实例（sslv.c:666-672）：*
struct sslv_config_data sslv_config = {
    .id = 0,
    .bits_per_word = 8,
    .sslv_pha = 1,
    .sslv_pol = 1,
    .loop_mode = 0,
};
`CMD_set_mode / CMD_set_bits` 修改的是这个全局结构体，`CMD_enable` 时把它整体写入硬件。

*struct sslv_device (sslv.h:32-39)*
struct sslv_device {
    struct sslv_config_data *config;  // 指向 sslv_config 全局实例
    int sslv_id;                      // = 0
    int status;                       // SSLV_IDLE / SSLV_BUSY
    int rx_flags;                     // 当前是否在 receive 路径
    int cb_flags;                     // 当前是否在 callback receive 路径
    struct mutex lock;                // 互斥锁
};

*struct jz_sslv_drv (sslv.c:34-63) — 最核心的运行态结构*
struct jz_sslv_drv {
    int id;              // 0
    int irq;             // IRQ_INTC_BASE + 6
    int is_enable;       // 模块参数控制是否初始化
    int is_finish;       // 初始化完成标志

    // GPIO 引脚编号
    int sslv_dt;         // PB29 (MOSI from slave 视角 = output)
    int sslv_dr;         // PB30 (MISO from slave 视角 = input)
    int sslv_clk;        // PB28
    int sslv_cs;         // PB31

    struct clk *clk;     // gate_ssislv 时钟

    // === 数据传输运行态 ===
    int tlen;            // 剩余待发送 word 数
    int rlen;            // 剩余待接收 word 数
    void *tbuff;         // 发送缓冲区当前位置
    void *rbuff;         // 接收缓冲区当前位置
    int rx_error;        // 接收错误码

    struct sslv_config_data *config;  // 当前配置
    unsigned int working_flag;       // 1=传输进行中, 0=空闲
    wait_queue_head_t waiter;        // wait_event 等待队列

    struct sslv_device dev;          // 嵌入的 sslv_device
    void (*sslv_receive_callback)(struct sslv_device *dev); // 回调接收
};
*关键点：tlen 和 rlen 的单位是 word 数（不是字节数）。tbuff 和 rbuff 按 bytes_per_word 步进。*

**ioctl 派发表**
CMD_enable   (110) → sslv_enable(&sslv_config)     // 写 CTRLR + 使能硬件
CMD_disable  (111) → sslv_disable(dev)             // 关硬件 → IDLE
CMD_receive  (112) → 分配内核缓冲 → 收数据 → 拷回用户
CMD_send     (113) → 直接发数据（未用内核缓冲，有 bug）
CMD_get_info (114) → 拷 sslv_config 到用户空间
CMD_set_mode (115) → 修改 sslv_config.sslv_pol/pha
CMD_set_bits (116) → 修改 sslv_config.bits_per_word

**接收数据流（核心路径）**
*CMD_receive 的完整链路：*

ioctl(CMD_receive, [user_buf, 8])
  │
  ├─ copy_from_user(array, arg, 16)         // 取出 [user_buf地址, len=8]
  │
  ├─ kmalloc(len)                           // 内核缓冲
  │
  ├─ sslv_receive(dev, kernel_buf, 8)       // 8 words
  │    │
  │    ├─ dev->rx_flags = 1                 // 标记接收中
  │    │
  │    └─ do_sslv_receive(dev, kernel_buf, 8)
  │         │
  │         ├─ sslv_hal_disable(0)          // SSIENR=0 → 清空 FIFO ⚠️
  │         ├─ 设 RXFTLR=7, IMR=0x18
  │         ├─ drv->rlen=8, working=1
  │         ├─ sslv_hal_enable(0)           // SSIENR=1
  │         │
  │         └─ wait_event_interruptible(waiter, !working_flag)
  │              │                           // 阻塞等中断
  │              │
  │              ├────────── 主机发送数据 ──────────
  │              │
  │              └─ ISR: sslv_intr_handler()
  │                   │
  │                   ├─ 读 ISR → rx_status=1
  │                   ├─ sslv_hal_clear_all_interrupt()
  │                   │
  │                   └─ sslv_rx_intr_default(dev)
  │                        │
  │                        ├─ FIFO_NUM = N
  │                        ├─ 读 min(rlen, N) 条到 rbuff
  │                        ├─ rlen -= N, rbuff += N*bytes
  │                        │
  │                        ├─ if rlen > 0: 更新阈值, 等下一次中断
  │                        └─ if rlen == 0:
  │                            关中断 → working_flag=0 → wake_up_all()
  │
  ├─ synchronize_irq()                      // 确保 ISR 不再用 kernel_buf
  ├─ copy_to_user(user_buf, kernel_buf, 8)
  └─ kfree(kernel_buf)

/* 使能传输 */
static void sslv_hal_enable_transfer(struct sslv_config_data *config)
{
